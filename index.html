function calculateAmount(){ const type = $("type").value; const amountInput = $("amount"); if(type === "sale"){ amountInput.readOnly = true; const quantity = Number($("quantity").value); const unitPrice = Number($("unitPrice").value); if($("quantity").value === "" || $("unitPrice").value === ""){ amountInput.value = ""; return; } amountInput.value = (quantity * unitPrice).toFixed(2); } else { amountInput.readOnly = false; } }
